# 日志配置
# 设置logcat缓存大小为20M
logcat -G 20M
# 开启eywa日志（设置日志级别）
setprop persist.xbh.hidl.loglevel 2
# 打开中间件日志
setprop persist.xbh.app_log.enable true
setprop persist.vendor.xbh.app_log.enable true
settings put global xbh_middleware_main_log_enable 1
# 开启中间件调用者打印
settings put global xbh_middleware_callLog_enable 1
# 开启API主日志
settings put global xbh_api_main_log_enable 1

# 日志抓取与保存
# 清除旧日志并抓取所有日志（含时间戳）保存到文件
logcat -c && logcat -b all -v time > /data/logcat.log 2>&1 &
# 下载日志到本地（示例）
adb pull /data/logcat.log .\
# U盘拷文件
- mount | grep -E "vfat|exfat" //查看挂载设备
- cp /data/logcat.log /mnt/media_rw/130A-3438 //拷贝
- ls /mnt/media_rw/130A-3438/ | grep logcat.log //验证
- umount /mnt/media_rw/130A-3438 //卸载

# 日志过滤
- logcat | grep -i -e "Skip network selection" -e "low signal strength"
