# 十六进制打印函数
HEX_LOG_FUNCTION := \
    void log_hex(const unsigned char* prefix, const unsigned char* cmd, int len) { \
        if (len > LOG_HEX_MAX_LEN) { \
            LOGI("too long hex data: %d > %d", len, LOG_HEX_MAX_LEN); \
            len = LOG_HEX_MAX_LEN; \
        } \
        char output[LOG_HEX_MAX_LEN * 3 + 1] = {0}; \
        const char hex[] = "0123456789ABCDEF"; \
        for (int i = 0; i < len; ++i) { \
            unsigned char byte = cmd[i]; \
            unsigned char high = (byte >> 4) & 0x0F; \
            unsigned char low = byte & 0x0F; \
            output[i * 3 + 0] = hex[high]; \
            output[i * 3 + 1] = hex[low]; \
            output[i * 3 + 2] = ' '; \
        } \
        LOGI("%s%s", prefix, output); \
    }
